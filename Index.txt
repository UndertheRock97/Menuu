    <script>
        let selectedFlavor = 'Plain';
        let mealAdded = false;
        const mealPrice = 450;

        const flavorButtons = document.querySelectorAll('.opt-btn');
        flavorButtons.forEach(btn => {
            btn.addEventListener('click', function() {
                flavorButtons.forEach(b => b.classList.remove('active'));
                this.classList.add('active');
                selectedFlavor = this.getAttribute('data-flavor');
            });
        });

        document.getElementById('add-meal-btn').addEventListener('click', function() {
            mealAdded = true;
            updateUI();
        });

        function updateUI() {
            const countText = document.getElementById('cart-count');
            const totalText = document.getElementById('cart-total');
            
            if (mealAdded) {
                countText.innerText = "1 item selected (Wok Whole Meal - " + selectedFlavor + ")";
                totalText.innerText = "KSh " + mealPrice;
            } else {
                countText.innerText = "0 items selected";
                totalText.innerText = "KSh 0";
            }
        }

        document.getElementById('checkout-btn').addEventListener('click', function() {
            if (!mealAdded) {
                alert("Please add your meal first.");
                return;
            }
            let text = "New Order - Under The Rock:\n- Wok Whole Meal (" + selectedFlavor + ") - KSh " + mealPrice + "\n\nTotal: KSh " + mealPrice + "\nPickup: Wanyee Rd, Dagoretti";
            window.location.href = `https://wa.me/254745552333?text=` + encodeURIComponent(text);
        });
    </script>
